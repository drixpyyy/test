-- ======================= UTILITY FUNCTIONS =======================
local function newPart(props)
    local p = Instance.new("Part")
    p.Size = props.size or Vector3.new(4, 4, 4)
    p.Position = props.pos or Vector3.new(0, 0, 0)
    p.Color = props.color or COLORS.Metal
    p.Material = props.mat or MAT.Metal
    p.Anchored = props.anchored ~= false
    p.CanCollide = props.collide ~= false
    p.Transparency = props.trans or 0
    p.Name = props.name or "Part"
    p.TopSurface = Enum.SurfaceType.Smooth
    p.BottomSurface = Enum.SurfaceType.Smooth
    p.Parent = props.parent or Skyforge
    if props.shape then p.Shape = props.shape end
    if props.cframe then p.CFrame = props.cframe end
    if props.reflect then p.Reflectance = props.reflect end
    if props.light then
        local l = Instance.new("PointLight", p)
        l.Color = props.lightColor or props.color or COLORS.NeonBlue
        l.Brightness = props.lightBright or 2
        l.Range = props.lightRange or 15
    end
    if props.glow then
        local sg = Instance.new("SurfaceLight", p)
        sg.Color = props.glowColor or props.color
        sg.Brightness = props.glowBright or 3
        sg.Range = props.glowRange or 20
        sg.Angle = props.glowAngle or 90
    end
    if props.decay then
        Debris:AddItem(p, props.decay)
    end
    return p
end

local function weld(p1, p2)
    local w = Instance.new("Weld")
    w.Part0 = p1
    w.Part1 = p2
    w.C0 = p1.CFrame:Inverse() * p2.CFrame
    w.Parent = p1
    return w
end

local function tween(obj, info, goals)
    return TS:Create(obj, TweenInfo.new(info.time or 0.5, info.ease or Enum.EasingStyle.Quad, info.dir or Enum.EasingDirection.Out), goals)
end

local function randomInRange(min, max)
    return min + math.random() * (max - min)
end

local function randomV3Range(minX, maxX, minY, maxY, minZ, maxZ)
    return Vector3.new(randomInRange(minX, maxX), randomInRange(minY, maxY), randomInRange(minZ, maxZ))
end

local function emitParticles(parent, color, count, size, speed, lifetime)
    local pe = Instance.new("ParticleEmitter", parent)
    pe.Color = ColorSequence.new(color or COLORS.NeonBlue)
    pe.Size = NumberSequence.new(size or 2)
    pe.Speed = NumberRange.new(speed or 10, (speed or 10) * 2)
    pe.Lifetime = NumberRange.new(lifetime or 1)
    pe.Rate = 0
    pe.SpreadAngle = Vector2.new(180, 180)
    pe:Emit(count or 10)
    Debris:AddItem(pe, (lifetime or 1) + 1)
    return pe
end

local function playSound(id, vol, pitch, parent)
    local s = Instance.new("Sound")
    s.SoundId = "rbxassetid://" .. id
    s.Volume = vol or 0.5
    s.PlaybackSpeed = pitch or 1
    s.Parent = parent or SoundService
    s:Play()
    Debris:AddItem(s, 5)
    return s
end

local function billboard(text, color, size, parent)
    local b = Instance.new("BillboardGui", parent)
    b.Size = UDim2.new(0, 200, 0, 50)
    b.StudsOffset = Vector3.new(0, 3, 0)
    b.AlwaysOnTop = true
    local l = Instance.new("TextLabel", b)
    l.Size = UDim2.new(1, 0, 1, 0)
    l.BackgroundTransparency = 1
    l.Text = text
    l.TextColor3 = color or COLORS.NeonBlue
    l.TextStrokeTransparency = 0.3
    l.TextStrokeColor3 = Color3.new(0, 0, 0)
    l.Font = Enum.Font.GothamBold
    l.TextSize = size or 18
    return b
end

local function createBasePlatform()

        parent = WorldFolder,
        glow = true,

        glowBright = 1,
        glowRange = 60
    })

    -- Edge ring with neon
    for i = 1, 32 do
        local angle = (i / 32) * math.pi * 2
        local pos = Vector3.new(math.cos(angle) * 145, 110, math.sin(angle) * 145)
        local edge = newPart({
            size = Vector3.new(6, 6, 6),
            pos = pos,
            color = COLORS.NeonBlue,
            mat = MAT.Base,
            name = "EdgeLight",
            parent = WorldFolder,
            light = true,
            lightColor = COLORS.NeonBlue,
            lightBright = 3,
            lightRange = 20
        })
    end

    -- Lower ring (gives depth)
    for i = 1, 24 do
        local angle = (i / 24) * math.pi * 2
        local pos = Vector3.new(math.cos(angle) * 135, 85, math.sin(angle) * 135)
        local lower = newPart({
            size = Vector3.new(8, 25, 8),
            pos = pos,
            color = COLORS.DarkMetal,
            mat = MAT.Metal,
            name = "PlatformSupport",
            parent = WorldFolder
        })
    end

    -- Central spire
    for i = 1, 12 do
        local spire = newPart({
            size = Vector3.new(12 - i, 8, 12 - i),
            pos = Vector3.new(0, 110 + i * 8, 0),
            color = COLORS.NeonPurple,
            mat = MAT.Base,
            name = "CentralSpire",
            parent = WorldFolder,
            light = true,
            lightColor = COLORS.NeonPurple,
            lightBright = 4,
            lightRange = 25
        })
    end

    -- Floating rocks around
    for i = 1, 20 do
        local angle = math.random() * math.pi * 2
        local dist = 170 + math.random() * 100
        local rock = newPart({
            size = Vector3.new(randomInRange(5, 15), randomInRange(5, 15), randomInRange(5, 15)),
            pos = Vector3.new(math.cos(angle) * dist, randomInRange(60, 140), math.sin(angle) * dist),
            color = COLORS.DarkMetal,
            mat = MAT.Metal,
            name = "FloatingRock",
            parent = WorldFolder
        })
        -- Slow rotation animation via tween
        local spin = tween(rock, {time = randomInRange(10, 30)}, {Orientation = Vector3.new(math.random()*360, math.random()*360, math.random()*360)})
        spin:Play()
    end

    return base
end

-- Building generation system
local function createSkyscraper(x, z, height, color, hasAntenna)
    local building = Instance.new("Model", BuildingsFolder)
    building.Name = "Skyscraper"

    -- Main tower
    local w = randomInRange(10, 20)
    local d = randomInRange(10, 20)
    local tower = newPart({
        size = Vector3.new(w, height, d),
        pos = Vector3.new(x, 110 + height/2, z),
        color = color or COLORS.Metal,
        mat = MAT.Metal,
        name = "Tower",
        parent = building
    })

    -- Windows (neon strips)
    local windowCount = math.floor(height / 8)
    for i = 1, windowCount do
        local side = math.random(1, 4)
        local wx, wz = x, z
        local ws = Vector3.new(0, 0, 0)
        if side == 1 then
            wx = x + w/2 + 0.2
            ws = Vector3.new(0.4, 4, d - 2)
        elseif side == 2 then
            wx = x - w/2 - 0.2
            ws = Vector3.new(0.4, 4, d - 2)
        elseif side == 3 then
            wz = z + d/2 + 0.2
            ws = Vector3.new(w - 2, 4, 0.4)
        else
            wz = z - d/2 - 0.2
            ws = Vector3.new(w - 2, 4, 0.4)
        end

        local winColor = ({COLORS.NeonBlue, COLORS.NeonPurple, COLORS.NeonPink, COLORS.NeonCyan, COLORS.NeonGreen})[math.random(1, 5)]
        newPart({
            size = ws,
            pos = Vector3.new(wx, 110 + i * 8, wz),
            color = winColor,
            mat = MAT.Base,
            name = "Window",
            parent = building,
            light = true,
            lightColor = winColor,
            lightBright = 1.5,
            lightRange = 10
        })
    end

    -- Top antenna
    if hasAntenna then
        for i = 1, 5 do
            newPart({
                size = Vector3.new(1, 8, 1),
                pos = Vector3.new(x, 110 + height + i * 8, z),
                color = COLORS.NeonOrange,
                mat = MAT.Base,
                name = "Antenna",
                parent = building,
                light = true,
                lightColor = COLORS.NeonOrange,
                lightBright = 5,
                lightRange = 15
            })
        end
        -- Beacon light
        local beacon = newPart({
            size = Vector3.new(3, 3, 3),
            pos = Vector3.new(x, 110 + height + 48, z),
            color = COLORS.Warning,
            mat = MAT.Base,
            name = "Beacon",
            parent = building
        })
        -- Blink animation
        spawn(function()
            while beacon.Parent do
                beacon.Transparency = 0
                wait(0.5)
                if not beacon.Parent then break end
                beacon.Transparency = 0.8
                wait(0.5)
            end
        end)
    end

    -- Roof detail
    newPart({
        size = Vector3.new(w + 4, 2, d + 4),
        pos = Vector3.new(x, 110 + height + 1, z),
        color = COLORS.DarkMetal,
        mat = MAT.Metal,
        name = "Roof",
        parent = building
    })

    return building
end

-- Bridge between platforms
local function createBridge(x1, z1, x2, z2, y)
    local dx, dz = x2 - x1, z2 - z1
    local dist = math.sqrt(dx*dx + dz*dz)
    local midX, midZ = (x1 + x2) / 2, (z1 + z2) / 2
    local angle = math.atan2(dz, dx)

    local bridge = newPart({
        size = Vector3.new(dist, 2, 8),
        pos = Vector3.new(midX, y, midZ),
        color = COLORS.DarkMetal,
        mat = MAT.Metal,
        name = "Bridge",
        parent = WorldFolder,
        cframe = CFrame.new(midX, y, midZ) * CFrame.Angles(0, -angle, 0)
    })

    -- Neon railing
    for side = -1, 1, 2 do
        local rail = newPart({
            size = Vector3.new(dist, 1, 0.5),
            pos = Vector3.new(0, 0, 0),
            color = COLORS.NeonBlue,
            mat = MAT.Base,
            name = "Rail",
            parent = WorldFolder
        })
        rail.CFrame = CFrame.new(midX, y + 3, midZ) * CFrame.Angles(0, -angle, 0) * CFrame.new(0, 0, side * 4)
    end

    return bridge
end

-- Decorative elements
local function createDecorations()
    -- Neon signs/billboards
    for i = 1, 8 do
        local angle = (i / 8) * math.pi * 2
        local pos = Vector3.new(math.cos(angle) * 100, 130, math.sin(angle) * 100)
        local sign = newPart({
            size = Vector3.new(12, 8, 1),
            pos = pos,
            color = ({COLORS.NeonPink, COLORS.NeonCyan, COLORS.NeonOrange, COLORS.NeonGreen})[math.random(1, 4)],
            mat = MAT.Base,
            name = "NeonSign",
            parent = WorldFolder,
            glow = true,
            glowBright = 5,
            glowRange = 30
        })
        sign.Orientation = Vector3.new(0, math.deg(angle) + 90, 0)
        billboard("SECTOR " .. i, COLORS.White, 14, sign)
    end

    -- Floating holographic rings
    for i = 1, 6 do
        local ring = newPart({
            size = Vector3.new(randomInRange(15, 30), 1, randomInRange(15, 30)),
            pos = randomV3Range(-120, 120, 130, 180, -120, 120),
            color = COLORS.NeonPurple,
            mat = MAT.Forcefield,
            name = "HoloRing",
            parent = WorldFolder,
            trans = 0.6,
            shape = Enum.PartType.Cylinder
        })
        ring.Orientation = Vector3.new(0, 0, 90)
        -- Float animation
        local origY = ring.Position.Y
        spawn(function()
            while ring.Parent do
                tween(ring, {time = 3, ease = Enum.EasingStyle.Sine, dir = Enum.EasingDirection.InOut}, {Position = Vector3.new(ring.Position.X, origY + 5, ring.Position.Z)}):Play()
                wait(3)
                if not ring.Parent then break end
                tween(ring, {time = 3, ease = Enum.EasingStyle.Sine, dir = Enum.EasingDirection.InOut}, {Position = Vector3.new(ring.Position.X, origY - 5, ring.Position.Z)}):Play()
                wait(3)
            end
        end)
    end

    -- Data pillars
    for i = 1, 12 do
        local angle = (i / 12) * math.pi * 2
        local pillar = newPart({
            size = Vector3.new(3, randomInRange(20, 40), 3),
            pos = Vector3.new(math.cos(angle) * randomInRange(40, 90), 110 + randomInRange(10, 30), math.sin(angle) * randomInRange(40, 90)),
            color = COLORS.NeonCyan,
            mat = MAT.Glass,
            name = "DataPillar",
            parent = WorldFolder,
            trans = 0.3,
            reflect = 0.5
        })
    end

    -- Energy orbs floating
    for i = 1, 15 do
        local orb = newPart({
            size = Vector3.new(2, 2, 2),
            pos = randomV3Range(-130, 130, 115, 200, -130, 130),
            color = COLORS.NeonGreen,
            mat = MAT.Base,
            name = "EnergyOrb",
            parent = WorldFolder,
            shape = Enum.PartType.Ball,
            light = true,
            lightColor = COLORS.NeonGreen,
            lightBright = 5,
            lightRange = 15
        })
        -- Orbit animation
        spawn(function()
            local t = 0
            local center = orb.Position
            local radius = randomInRange(5, 15)
            local speed = randomInRange(0.5, 2)
            while orb.Parent do
                t = t + 0.02
                orb.Position = center + Vector3.new(math.cos(t * speed) * radius, math.sin(t * 0.5) * 5, math.sin(t * speed) * radius)
                wait(0.03)
            end
        end)
    end
end

-- Fire weapon
local function fireWeapon()
    local now = tick()
    if now - lastFire < fireRate then return end
    lastFire = now

    local camCF = cam.CFrame
    local startPos = camCF.Position
    local direction = camCF.LookVector

    -- Barrel flash
    local flash = newPart({
        size = Vector3.new(1.5, 1.5, 1.5),
        pos = startPos + direction * 3,
        color = COLORS.NeonBlue,
        mat = MAT.Base,
        name = "MuzzleFlash",
        parent = EffectsFolder,
        anchored = true,
        collide = false,
        trans = 0.2
    })
    tween(flash, {time = 0.05}, {Size = Vector3.new(0.5, 0.5, 0.5), Transparency = 1}):Play()
    Debris:AddItem(flash, 0.1)

    -- Laser beam
    local beam = Instance.new("Part", EffectsFolder)
    beam.Size = Vector3.new(0.15, 0.15, 500)
    beam.CFrame = CFrame.new(startPos + direction * 250, startPos + direction * 500)
    beam.Anchored = true
    beam.CanCollide = false
    beam.Color = COLORS.NeonBlue
    beam.Material = MAT.Base
    beam.Transparency = 0.3
    tween(beam, {time = 0.1}, {Transparency = 1}):Play()
    Debris:AddItem(beam, 0.1)

    -- Screen shake
    local origCF = cam.CFrame
    cam.CFrame = origCF * CFrame.new(randomInRange(-0.1, 0.1), randomInRange(-0.1, 0.1), 0)
    delay(0.03, function() cam.CFrame = origCF end)

    -- Hit detection
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {char, weaponModel, EffectsFolder, ProjectilesFolder}
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    local result = workspace:Raycast(startPos, direction * 500, rayParams)

    if result then
        local hitPos = result.Position
        local hitInst = result.Instance

        -- Hit effect
        local hitExplosion = newPart({
            size = Vector3.new(2, 2, 2),
            pos = hitPos,
            color = COLORS.NeonBlue,
            mat = MAT.Base,
            name = "HitEffect",
            parent = EffectsFolder,
            anchored = true,
            collide = false,
            trans = 0.3,
            shape = Enum.PartType.Ball
        })
        tween(hitExplosion, {time = 0.3}, {Size = Vector3.new(6, 6, 6), Transparency = 1}):Play()
        Debris:AddItem(hitExplosion, 0.3)
        emitParticles(hitExplosion, COLORS.NeonBlue, 15, 1, 15, 0.5)

        -- Check if enemy hit
        local enemyModel = hitInst:FindFirstAncestorOfClass("Model")
        if enemyModel and enemyModel.Parent == EnemiesFolder then
            local hpVal = enemyModel:FindFirstChild("HP")
            if hpVal then
                hpVal.Value = hpVal.Value - weaponDamage
                -- Damage number
                local dmgBillboard = billboard("-" .. weaponDamage, COLORS.NeonOrange, 20, hitInst)
                dmgBillboard.StudsOffset = Vector3.new(randomInRange(-2, 2), 4, randomInRange(-2, 2))
                tween(dmgBillboard, {time = 0.8}, {StudsOffset = Vector3.new(0, 8, 0)}):Play()
                Debris:AddItem(dmgBillboard, 0.8)

                -- Enemy flash red
                for _, part in ipairs(enemyModel:GetDescendants()) do
                    if part:IsA("BasePart") then
                        local origColor = part.Color
                        part.Color = COLORS.Warning
                        delay(0.1, function()
                            if part.Parent then part.Color = origColor end
                        end)
                    end
                end

                if hpVal.Value <= 0 then
                    -- Enemy death
                    for _, part in ipairs(enemyModel:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.Anchored = false
                            part.CanCollide = true
                            part.Velocity = Vector3.new(randomInRange(-30, 30), randomInRange(20, 50), randomInRange(-30, 30))
                            part.RotVelocity = Vector3.new(randomInRange(-10, 10), randomInRange(-10, 10), randomInRange(-10, 10))
                            emitParticles(part, COLORS.NeonOrange, 10, 1, 15, 1)
                        end
                    end
                    emitParticles(enemyModel:FindFirstChild("HumanoidRootPart") or enemyModel:FindFirstChildWhichIsA("Part"), COLORS.NeonOrange, 40, 3, 20, 2)
                    playSound(5801257795, 0.8, 1, humRoot)

                    -- Score popup
                    addScore(100)
                    local scoreBillboard = billboard("+100", COLORS.NeonGreen, 24, humRoot)
                    scoreBillboard.StudsOffset = Vector3.new(0, 6, 0)
                    tween(scoreBillboard, {time = 1}, {StudsOffset = Vector3.new(0, 12, 0)}):Play()
                    Debris:AddItem(scoreBillboard, 1)

                    Debris:AddItem(enemyModel, 3)
                end
            end
        end
    end

    playSound(3932505916, 0.15, randomInRange(1.8, 2.2), humRoot)
end

local enemiesAlive = 0
local maxEnemies = 12

local function spawnEnemy(x, y, z, type)
    if enemiesAlive >= maxEnemies then return end
    enemiesAlive = enemiesAlive + 1

    local enemy = Instance.new("Model", EnemiesFolder)
    enemy.Name = "EnemyDrone"

    local hpVal = Instance.new("IntValue", enemy)
    hpVal.Name = "HP"
    hpVal.Value = (type == "heavy" and 150) or (type == "fast" and 50) or 100

    local maxHp = hpVal.Value

    -- Body
    local body = newPart({
        size = (type == "heavy" and Vector3.new(5, 5, 5)) or (type == "fast" and Vector3.new(2.5, 2.5, 2.5)) or Vector3.new(3.5, 3.5, 3.5),
        pos = Vector3.new(x, y, z),
        color = (type == "heavy" and COLORS.Warning) or (type == "fast" and COLORS.NeonGreen) or COLORS.NeonPink,
        mat = MAT.Metal,
        name = "Body",
        parent = enemy,
        shape = Enum.PartType.Ball
    })

    -- Eye
    local eye = newPart({
        size = Vector3.new(1, 1, 0.5),
        pos = Vector3.new(x, y, z - 2),
        color = COLORS.NeonRed or COLORS.Warning,
        mat = MAT.Base,
        name = "Eye",
        parent = enemy
    })
    weld(body, eye)

    -- Light
    local elight = Instance.new("PointLight", body)
    elight.Color = body.Color
    elight.Brightness = 3
    elight.Range = 20

    -- Health bar
    local hpGui = Instance.new("BillboardGui", body)
    hpGui.Size = UDim2.new(0, 80, 0, 8)
    hpGui.StudsOffset = Vector3.new(0, 4, 0)
    hpGui.AlwaysOnTop = true
    local hpBg = Instance.new("Frame", hpGui)
    hpBg.Size = UDim2.new(1, 0, 1, 0)
    hpBg.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    hpBg.BorderSizePixel = 0
    local hpFill = Instance.new("Frame", hpBg)
    hpFill.Size = UDim2.new(1, 0, 1, 0)
    hpFill.BackgroundColor3 = COLORS.NeonGreen
    hpFill.BorderSizePixel = 0

    -- AI Behavior
    spawn(function()
        local speed = (type == "fast" and 18) or (type == "heavy" and 6) or 12
        local attackRange = 15
        local attackCooldown = 0
        local idleOffset = Vector3.new(randomInRange(-20, 20), randomInRange(-5, 10), randomInRange(-20, 20))
        local startPos = Vector3.new(x, y, z)

        while enemy.Parent and hpVal.Value > 0 do
            local eChar = player.Character
            if eChar and eChar:FindFirstChild("HumanoidRootPart") then
                local targetPos = eChar.HumanoidRootPart.Position
                local myPos = body.Position
                local dist = (targetPos - myPos).Magnitude

                -- Update health bar
                hpFill.Size = UDim2.new(hpVal.Value / maxHp, 0, 1, 0)
                hpFill.BackgroundColor3 = hpVal.Value > maxHp * 0.5 and COLORS.NeonGreen or (hpVal.Value > maxHp * 0.25 and COLORS.Warning or COLORS.NeonPink)

                if dist > 200 then
                    -- Too far, return to patrol
                    local patrolPos = startPos + idleOffset
                    local dir = (patrolPos - myPos).Unit
                    body.Position = body.Position + dir * speed * 0.03
                    body.CFrame = CFrame.new(body.Position, body.Position + dir)
                elseif dist > attackRange then
                    -- Chase player
                    local dir = (targetPos - myPos).Unit
                    body.Position = body.Position + dir * speed * 0.03
                    body.CFrame = CFrame.new(body.Position, targetPos)

                    -- Movement trail
                    if math.random() > 0.7 then
                        local trail = newPart({
                            size = Vector3.new(0.3, 0.3, 0.3),
                            pos = body.Position,
                            color = body.Color,
                            mat = MAT.Base,
                            name = "Trail",
                            parent = EffectsFolder,
                            anchored = true,
                            collide = false,
                            trans = 0.5,
                            shape = Enum.PartType.Ball
                        })
                        tween(trail, {time = 0.5}, {Size = Vector3.new(0, 0, 0), Transparency = 1}):Play()
                        Debris:AddItem(trail, 0.5)
                    end
                else
                    -- Attack
                    body.CFrame = CFrame.new(body.Position, targetPos)
                    if tick() - attackCooldown > 1 then
                        attackCooldown = tick()

                        -- Shoot at player
                        local proj = newPart({
                            size = Vector3.new(0.8, 0.8, 0.8),
                            pos = body.Position,
                            color = COLORS.Warning,
                            mat = MAT.Base,
                            name = "EnemyProjectile",
                            parent = ProjectilesFolder,
                            anchored = true,
                            collide = false,
                            shape = Enum.PartType.Ball,
                            light = true,
                            lightColor = COLORS.Warning,
                            lightBright = 5,
                            lightRange = 10
                        })

                        local projDir = (targetPos - body.Position).Unit
                        local projSpeed = 40
                        spawn(function()
                            local travelled = 0
                            while proj.Parent and travelled < 300 do
                                proj.Position = proj.Position + projDir * projSpeed * 0.03
                                travelled = travelled + projSpeed * 0.03

                                -- Check player hit
                                if eChar and eChar:FindFirstChild("HumanoidRootPart") then
                                    if (proj.Position - eChar.HumanoidRootPart.Position).Magnitude < 4 then
                                        -- Hit player
                                        takeDamage(10)
                                        emitParticles(proj, COLORS.Warning, 15, 2, 15, 0.5)
                                        playSound(5801250166, 0.5, 1, eChar.HumanoidRootPart)
                                        proj:Destroy()
                                        return
                                    end
                                end

                                -- Sparkle
                                local spark = newPart({
                                    size = Vector3.new(0.2, 0.2, 0.2),
                                    pos = proj.Position,
                                    color = COLORS.Warning,
                                    mat = MAT.Base,
                                    name = "Spark",
                                    parent = EffectsFolder,
                                    anchored = true,
                                    collide = false,
                                    trans = 0.3
                                })
                                tween(spark, {time = 0.2}, {Transparency = 1}):Play()
                                Debris:AddItem(spark, 0.2)

                                wait(0.03)
                            end
                            if proj.Parent then
                                emitParticles(proj, COLORS.Warning, 10, 1, 10, 0.3)
                                proj:Destroy()
                            end
                        end)

                        playSound(3932505916, 0.2, 0.8, body)
                    end
                end
            end
            wait(0.03)
        end
        enemiesAlive = enemiesAlive - 1
    end)

    return enemy
end

-- ======================= GAME STATE =======================
local Score = 0
local Combo = 0
local MaxHealth = 200
local Health = 200
local Shield = 100
local lastKillTime = 0

local function addScore(amount)
    local now = tick()
    if now - lastKillTime < 2 then
        Combo = Combo + 1
        amount = amount * (1 + Combo * 0.5)
    else
        Combo = 1
    end
    lastKillTime = now
    Score = Score + math.floor(amount)
end

local function takeDamage(amount)
    if Shield > 0 then
        Shield = math.max(0, Shield - amount)
        Health = Health - (amount * 0.2)
    else
        Health = Health - amount
    end

    -- Screen flash red
    local flash = Instance.new("ScreenGui", player:WaitForChild("PlayerGui"))
    flash.Name = "DamageFlash"
    local fFrame = Instance.new("Frame", flash)
    fFrame.Size = UDim2.new(1, 0, 1, 0)
    fFrame.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
    fFrame.BackgroundTransparency = 0.6
    fFrame.BorderSizePixel = 0
    tween(fFrame, {time = 0.3}, {BackgroundTransparency = 1}):Play()
    Debris:AddItem(flash, 0.3)

    if Health <= 0 then
        Health = MaxHealth
        Shield = 100
        humRoot.CFrame = CFrame.new(0, 130, 0)
        emitParticles(humRoot, COLORS.NeonBlue, 50, 3, 20, 2)
        playSound(5801637099, 1, 0.8, humRoot)

        -- Death popup
        local deathGui = Instance.new("ScreenGui", player.PlayerGui)
        deathGui.Name = "DeathMessage"
        local dText = Instance.new("TextLabel", deathGui)
        dText.Size = UDim2.new(0.4, 0, 0.1, 0)
        dText.Position = UDim2.new(0.3, 0, 0.4, 0)
        dText.BackgroundTransparency = 1
        dText.Text = "SYSTEM REBOOTED"

        dText.TextStrokeColor3 = Color3.new(0, 0, 0)

        dText.TextSize = 40
        tween(dText, {time = 2}, {TextTransparency = 1, TextStrokeTransparency = 1}):Play()
        Debris:AddItem(deathGui, 2)
    end
end

local collectibleCount = 0
local maxCollectibles = 15

local function spawnCollectible()
    if collectibleCount >= maxCollectibles then return end
    collectibleCount = collectibleCount + 1

    local pos = randomV3Range(-130, 130, 115, 200, -130, 130)
    local types = {"health", "shield", "score", "ammo"}
    local cType = types[math.random(1, #types)]
    local color = (cType == "health" and COLORS.NeonPink) or (cType == "shield" and COLORS.NeonBlue) or (cType == "score" and COLORS.NeonGreen) or COLORS.NeonOrange

    local orb = newPart({
        size = Vector3.new(2, 2, 2),
        pos = pos,
        color = color,
        mat = MAT.Base,
        name = "Collectible_" .. cType,
        parent = ItemsFolder,
        shape = Enum.PartType.Ball,
        light = true,
        lightColor = color,
        lightBright = 5,
        lightRange = 15
    })

    -- Float animation
    spawn(function()
        local origY = orb.Position.Y
        local t = 0
        while orb.Parent do
            t = t + 0.03
            orb.Position = Vector3.new(orb.Position.X, origY + math.sin(t * 2) * 3, orb.Position.Z)
            orb.Rotation = Vector3.new(0, t * 50, 0)
            wait(0.03)
        end
    end)

    -- Collection check
    spawn(function()
        while orb.Parent do
            local eChar = player.Character
            if eChar and eChar:FindFirstChild("HumanoidRootPart") then
                if (orb.Position - eChar.HumanoidRootPart.Position).Magnitude < 5 then
                    if cType == "health" then
                        Health = math.min(MaxHealth, Health + 50)
                    elseif cType == "shield" then
                        Shield = math.min(100, Shield + 50)
                    elseif cType == "score" then
                        addScore(250)
                    end

                    -- Collect effect
                    emitParticles(orb, color, 30, 2, 20, 1)
                    playSound(5801630167, 0.5, 1.5, orb)
                    collectibleCount = collectibleCount - 1
                    orb:Destroy()
                    return
                end
            end
            wait(0.1)
        end
    end)
end

-- Minimap dots storage
local mmDots = {}

local function getMMDot(color, size)
    local dot = Instance.new("Frame")
    dot.Size = UDim2.new(0, size or 4, 0, size or 4)
    dot.BackgroundColor3 = color
    dot.BorderSizePixel = 0
    local corner = Instance.new("UICorner", dot)
    corner.CornerRadius = UDim.new(1, 0)
    return dot
end

-- Update minimap
spawn(function()
    while minimapGui.Parent do
        -- Clear old dots
        for _, dot in ipairs(mmDots) do
            dot:Destroy()
        end
        mmDots = {}

        local playerPos = humRoot.Position
        local mapScale = 0.4 -- pixels per stud

        -- Player dot (center)
        local playerDot = getMMDot(COLORS.NeonGreen, 6)
        playerDot.Position = UDim2.new(0.5, -3, 0.5, -3)
        playerDot.Parent = mmContainer
        table.insert(mmDots, playerDot)

        -- Enemy dots
        for _, enemy in ipairs(EnemiesFolder:GetChildren()) do
            if enemy:FindFirstChild("Body") then
                local ep = enemy.Body.Position
                local relX = (ep.X - playerPos.X) * mapScale
                local relZ = (ep.Z - playerPos.Z) * mapScale
                if math.abs(relX) < 75 and math.abs(relZ) < 75 then
                    local dot = getMMDot(COLORS.Warning, 4)
                    dot.Position = UDim2.new(0.5, relX - 2, 0.5, relZ - 2)
                    dot.Parent = mmContainer
                    table.insert(mmDots, dot)
                end
            end
        end

        -- Collectible dots
        for _, item in ipairs(ItemsFolder:GetChildren()) do
            if item:IsA("BasePart") then
                local ip = item.Position
                local relX = (ip.X - playerPos.X) * mapScale
                local relZ = (ip.Z - playerPos.Z) * mapScale
                if math.abs(relX) < 75 and math.abs(relZ) < 75 then
                    local dot = getMMDot(COLORS.NeonCyan, 3)
                    dot.Position = UDim2.new(0.5, relX - 1.5, 0.5, relZ - 1.5)
                    dot.Parent = mmContainer
                    table.insert(mmDots, dot)
                end
            end
        end

        wait(0.2)
    end
end)

local teleporterPositions = {
    {pos = Vector3.new(0, 115, 0), name = "CENTER"},
    {pos = Vector3.new(100, 140, 0), name = "SECTOR A"},
    {pos = Vector3.new(-100, 130, 0), name = "SECTOR B"},
    {pos = Vector3.new(0, 150, 100), name = "SECTOR C"},
    {pos = Vector3.new(0, 125, -100), name = "SECTOR D"},
    {pos = Vector3.new(80, 180, 80), name = "SKY FORGE"},
    {pos = Vector3.new(-80, 160, -80), name = "VOID NEXUS"}
}

for i, tp in ipairs(teleporterPositions) do
    -- Teleporter pad
    local pad = newPart({
        size = Vector3.new(8, 1, 8),
        pos = tp.pos,
        color = COLORS.NeonPurple,
        mat = MAT.Base,
        name = "Teleporter_" .. tp.name,
        parent = WorldFolder,
        glow = true,
        glowColor = COLORS.NeonPurple,
        glowBright = 5,
        glowRange = 25
    })

    -- Label
    local tpLabel = billboard(tp.name, COLORS.NeonPurple, 16, pad)
    tpLabel.StudsOffset = Vector3.new(0, 4, 0)

    -- Particles
    local pe = Instance.new("ParticleEmitter", pad)
    pe.Color = ColorSequence.new(COLORS.NeonPurple)
    pe.Size = NumberSequence.new(1)
    pe.Speed = NumberRange.new(5, 10)
    pe.Lifetime = NumberRange.new(1, 2)
    pe.Rate = 10
    pe.SpreadAngle = Vector2.new(0, 0)
    pe.VelocityInheritance = 0

    -- Activation
    spawn(function()
        while pad.Parent do
            local eChar = player.Character
            if eChar and eChar:FindFirstChild("HumanoidRootPart") then
                if (pad.Position - eChar.HumanoidRootPart.Position).Magnitude < 6 then
                    -- Teleport to random other pad
                    local target = teleporterPositions[math.random(1, #teleporterPositions)]
                    if target.pos ~= tp.pos then
                        eChar.HumanoidRootPart.CFrame = CFrame.new(target.pos + Vector3.new(0, 5, 0))
                        emitParticles(eChar.HumanoidRootPart, COLORS.NeonPurple, 50, 3, 25, 2)
                        playSound(5801637099, 0.8, 1.2, eChar.HumanoidRootPart)
                        wait(2)
                    end
                end
            end
            wait(0.2)
        end
    end)
end

-- Lightning effects in the sky
spawn(function()
    while true do
        wait(randomInRange(3, 8))
        local start = randomV3Range(-200, 200, 250, 350, -200, 200)
        local end_ = randomV3Range(-100, 100, 100, 150, -100, 100)

        -- Bolt segments
        local segments = 8
        local current = start
        for i = 1, segments do
            local nextPos = current + (end_ - start) / segments + randomV3Range(-15, 15, -5, 5, -15, 15)
            local seg = newPart({
                size = Vector3.new(0.5, (nextPos - current).Magnitude, 0.5),
                pos = (current + nextPos) / 2,
                color = COLORS.NeonBlue,
                mat = MAT.Base,
                name = "Lightning",
                parent = EffectsFolder,
                anchored = true,
                collide = false,
                trans = 0.1,
                cframe = CFrame.lookAt(current, nextPos) * CFrame.Angles(math.pi/2, 0, 0)
            })
            tween(seg, {time = 0.2}, {Transparency = 1}):Play()
            Debris:AddItem(seg, 0.2)
            current = nextPos
        end

        -- Flash lighting
        Lighting.Brightness = 4
        delay(0.05, function()
            local restore = tween(Lighting, {time = 0.3}, {Brightness = 1.5})
            restore:Play()
        end)

        playSound(5801250166, 0.3, randomInRange(0.8, 1.2), humRoot)
    end
end)

spawn(function()
    local t = 20 -- Start at night
    while true do
        t = t + 0.005
        if t > 24 then t = 0 end
        Lighting.ClockTime = t

        -- Adjust colors based on time
        if t > 6 and t < 18 then
            -- Day
            local progress = (t - 6) / 12
            Lighting.Ambient = Color3.fromRGB(30 + progress * 50, 30 + progress * 60, 60 + progress * 40)
            atmosphere.Color = Color3.fromRGB(15 + progress * 40, 15 + progress * 50, 50 + progress * 30)
        else
            -- Night
            Lighting.Ambient = Color3.fromRGB(30, 30, 60)
            atmosphere.Color = Color3.fromRGB(15, 15, 50)
        end

        wait(0.5)
    end
end)

-- ======================= SECTOR DISCOVERY =======================
local discoveredSectors = {}
spawn(function()
    while true do
        wait(1)
        local pos = humRoot.Position
        local dist = math.sqrt(pos.X * pos.X + pos.Z * pos.Z)

        local sectorName
        if dist < 40 then
            sectorName = "CENTER HUB"
        elseif dist < 90 then
            sectorName = "INNER RING"
        elseif dist < 140 then
            sectorName = "OUTER RING"
        else
            sectorName = "FRONTIER"
        end

        if not discoveredSectors[sectorName] then
            discoveredSectors[sectorName] = true

            -- Discovery popup
            local discGui = Instance.new("ScreenGui", playerGui)
            discGui.Name = "Discovery"
            local discLabel = Instance.new("TextLabel", discGui)
            discLabel.Size = UDim2.new(0.4, 0, 0.08, 0)
            discLabel.Position = UDim2.new(0.3, 0, 0.2, 0)
            discLabel.BackgroundTransparency = 1
            discLabel.Text = "SECTOR DISCOVERED: " .. sectorName
            discLabel.TextColor3 = COLORS.NeonGreen
            discLabel.TextStrokeTransparency = 0
            discLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
            discLabel.Font = Enum.Font.GothamBlack
            discLabel.TextSize = 28

            tween(discLabel, {time = 0.5}, {Position = UDim2.new(0.3, 0, 0.15, 0)}):Play()
            delay(2, function()
                local fade = tween(discLabel, {time = 1}, {TextTransparency = 1, TextStrokeTransparency = 1})
                fade:Play()
                fade.Completed:Connect(function() discGui:Destroy() end)
            end)

            addScore(500)
            emitParticles(humRoot, COLORS.NeonGreen, 30, 2, 15, 1.5)
            playSound(5801637099, 0.6, 1, humRoot)
        end
    end
end)
